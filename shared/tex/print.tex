% Print preamble shared by every book.
% Trim size is set in shared/quarto-common.yml (7.5 x 9.25 in, the size of
% most O'Reilly-style technical books). Change it there, in one place.

\usepackage{fontspec}
\usepackage{microtype}
\usepackage{xcolor}
\usepackage{fvextra}

% Code: smaller, wraps long lines, no colour dependence so greyscale printing
% stays readable. Pandoc's Shaded and Highlighting environments are adjusted.
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  commandchars=\\\{\},
  breaklines,
  breakanywhere,
  fontsize=\footnotesize
}

% Print shows every language of an example one after another. The HTML tabset
% headings ("R", "Python", "JavaScript") become run-in labels.
\usepackage{titlesec}
\titleformat{\paragraph}[runin]{\bfseries\sffamily}{}{0pt}{}[\ ]

% Headers and folios.
\usepackage{scrlayer-scrpage}
\pagestyle{scrheadings}
\clearpairofpagestyles
\ohead{\headmark}
\ofoot*{\pagemark}
\cfoot*{}
\automark[section]{chapter}

% Avoid widows and orphans in the printed book.
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000
